# ============================================================
# CONFIGURATION
# ============================================================

ROBOT_IP = '10.12.34.2'

# ------------------------------------------------------------
# LiDAR
# ------------------------------------------------------------

MAX_DISTANCE_METERS = 5.0
MIN_DISTANCE_METERS = 0.05

# Maximum distance between consecutive LiDAR points
# before they are treated as a separate surface.
CONNECT_DISTANCE = 0.30

# Maximum number of points used for mapping each scan.
# Reducing this improves performance.
MAX_SCAN_POINTS = 360


# ------------------------------------------------------------
# MAP
# ------------------------------------------------------------

MAP_SIZE_METERS = 20.0
GRID_RESOLUTION = 0.05       # 5 cm per cell

MAP_CELLS = int(
    MAP_SIZE_METERS / GRID_RESOLUTION
)

# Log-odds occupancy values
FREE_UPDATE = -0.25
OCCUPIED_UPDATE = 0.85

LOG_ODDS_MIN = -5.0
LOG_ODDS_MAX = 5.0


# ------------------------------------------------------------
# ROBOT POSE / LOCALISATION
# ------------------------------------------------------------
#
# CHANGE THESE TO MATCH YOUR ROBOT'S ACTUAL
# NETWORKTABLES STRUCTURE.
#
# Example expected values:
#
# X = metres
# Y = metres
# Heading = degrees
#
# Heading convention used here:
#
#       0°
#        ↑
#        |
#  90° ← ● → -90°
#        |
#       180°
#
# Positive heading = clockwise
#
# ------------------------------------------------------------

POSE_TABLE_NAME = "Drive"

POSE_X_KEY = "PoseX"
POSE_Y_KEY = "PoseY"
POSE_HEADING_KEY = "PoseHeading"


# ------------------------------------------------------------
# LOCAL OBSTACLE WARNING
# ------------------------------------------------------------

OBSTACLE_WARNING_DISTANCE = 0.75


# ============================================================
# THREAD SAFETY
# ============================================================

data_lock = threading.Lock()


# ============================================================
# NETWORK TABLE DATA
# ============================================================

angles_1 = np.array([])
distances_1 = np.array([])

angles_2 = np.array([])
distances_2 = np.array([])


# Latest robot pose
robot_x = 0.0
robot_y = 0.0
robot_heading = 0.0


# Used to detect new scans
scan_update_counter = 0
last_mapped_scan = -1


# ============================================================
# GLOBAL MAP
# ============================================================

occupancy_grid = np.zeros(
    (MAP_CELLS, MAP_CELLS),
    dtype=np.float32
)


# Robot trajectory
robot_path_x = []
robot_path_y = []


# ============================================================
# NETWORKTABLE CALLBACK
# ============================================================

def value_changed_callback(table, key, value, isNew):

    global angles_1
    global distances_1
    global angles_2
    global distances_2

    global robot_x
    global robot_y
    global robot_heading

    global scan_update_counter

    with data_lock:

        # ----------------------------------------------------
        # LiDAR
        # ----------------------------------------------------

        if key == 'ScanAngles_Part1':
            angles_1 = np.asarray(value, dtype=float)

        elif key == 'ScanDistances_Part1':
            distances_1 = np.asarray(value, dtype=float)

        elif key == 'ScanAngles_Part2':
            angles_2 = np.asarray(value, dtype=float)

        elif key == 'ScanDistances_Part2':
            distances_2 = np.asarray(value, dtype=float)

            # Treat Part 2 distance update as completion
            # of one scan.
            scan_update_counter += 1


        # ----------------------------------------------------
        # ROBOT POSE
        # ----------------------------------------------------

        elif key == POSE_X_KEY:
            robot_x = float(value)

        elif key == POSE_Y_KEY:
            robot_y = float(value)

        elif key == POSE_HEADING_KEY:
            robot_heading = float(value)


# ============================================================
# GET COMPLETE LIDAR SCAN
# ============================================================

def get_combined_scan():

    with data_lock:

        a1 = angles_1.copy()
        d1 = distances_1.copy()

        a2 = angles_2.copy()
        d2 = distances_2.copy()

    # --------------------------------------------------------
    # Validate parts
    # --------------------------------------------------------

    n1 = min(len(a1), len(d1))
    n2 = min(len(a2), len(d2))

    if n1 == 0 and n2 == 0:
        return np.array([]), np.array([])

    angles = np.concatenate([
        a1[:n1],
        a2[:n2]
    ])

    distances = np.concatenate([
        d1[:n1],
        d2[:n2]
    ])

    # --------------------------------------------------------
    # Remove invalid measurements
    # --------------------------------------------------------

    valid = (
        np.isfinite(angles)
        & np.isfinite(distances)
    )

    angles = angles[valid]
    distances = distances[valid]

    # --------------------------------------------------------
    # Sort by angle
    # --------------------------------------------------------

    order = np.argsort(angles)

    angles = angles[order]
    distances = distances[order]

    return angles, distances


# ============================================================
# COORDINATE CONVERSION
# ============================================================

def lidar_to_local(angles_deg, distances_mm):

    distances_m = distances_mm / 1000.0

    # 0° = forward
    #
    # Positive X = right
    # Positive Y = forward

    angles_rad = np.radians(angles_deg)

    local_x = (
        distances_m
        * np.sin(angles_rad)
    )

    local_y = (
        distances_m
        * np.cos(angles_rad)
    )

    return local_x, local_y


# ============================================================
# LOCAL → GLOBAL COORDINATES
# ============================================================

def local_to_global(
    local_x,
    local_y,
    x_robot,
    y_robot,
    heading_deg
):

    heading = np.radians(heading_deg)

    # Heading is defined clockwise from forward (+Y)
    #
    # Forward vector:
    #     (sin(h), cos(h))
    #
    # Right vector:
    #     (cos(h), -sin(h))

    global_x = (
        x_robot
        + local_x * np.cos(heading)
        + local_y * np.sin(heading)
    )

    global_y = (
        y_robot
        - local_x * np.sin(heading)
        + local_y * np.cos(heading)
    )

    return global_x, global_y


# ============================================================
# WORLD → GRID
# ============================================================

def world_to_grid(x, y):

    map_min = -MAP_SIZE_METERS / 2

    grid_x = (
        (x - map_min)
        / GRID_RESOLUTION
    ).astype(int)

    grid_y = (
        (y - map_min)
        / GRID_RESOLUTION
    ).astype(int)

    return grid_x, grid_y


# ============================================================
# GRID VALIDATION
# ============================================================

def valid_grid_cell(x, y):

    return (
        0 <= x < MAP_CELLS
        and
        0 <= y < MAP_CELLS
    )


# ============================================================
# BRESENHAM LINE
# ============================================================

def bresenham(x0, y0, x1, y1):

    points = []

    dx = abs(x1 - x0)
    dy = abs(y1 - y0)

    sx = 1 if x0 < x1 else -1
    sy = 1 if y0 < y1 else -1

    err = dx - dy

    while True:

        points.append((x0, y0))

        if x0 == x1 and y0 == y1:
            break

        e2 = 2 * err

        if e2 > -dy:

            err -= dy
            x0 += sx

        if e2 < dx:

            err += dx
            y0 += sy

    return points


# ============================================================
# UPDATE OCCUPANCY MAP
# ============================================================

def update_occupancy_map(
    local_x,
    local_y,
    x_robot,
    y_robot,
    heading
):

    global occupancy_grid

    # --------------------------------------------------------
    # Convert LiDAR points to world coordinates
    # --------------------------------------------------------

    world_x, world_y = local_to_global(
        local_x,
        local_y,
        x_robot,
        y_robot,
        heading
    )

    # Robot's grid position
    robot_grid_x, robot_grid_y = world_to_grid(
        np.array([x_robot]),
        np.array([y_robot])
    )

    robot_grid_x = int(robot_grid_x[0])
    robot_grid_y = int(robot_grid_y[0])

    if not valid_grid_cell(
        robot_grid_x,
        robot_grid_y
    ):
        return


    # --------------------------------------------------------
    # Convert obstacle endpoints to grid coordinates
    # --------------------------------------------------------

    obstacle_grid_x, obstacle_grid_y = world_to_grid(
        world_x,
        world_y
    )


    # --------------------------------------------------------
    # Raycast every LiDAR beam
    # --------------------------------------------------------

    for gx, gy in zip(
        obstacle_grid_x,
        obstacle_grid_y
    ):

        if not valid_grid_cell(gx, gy):
            continue

        # Find cells between robot and obstacle
        ray = bresenham(
            robot_grid_x,
            robot_grid_y,
            int(gx),
            int(gy)
        )

        if len(ray) < 2:
            continue

        # ----------------------------------------------------
        # Mark cells along ray as FREE
        # ----------------------------------------------------

        for cell_x, cell_y in ray[:-1]:

            if valid_grid_cell(
                cell_x,
                cell_y
            ):

                occupancy_grid[
                    cell_y,
                    cell_x
                ] += FREE_UPDATE


        # ----------------------------------------------------
        # Final cell = obstacle
        # ----------------------------------------------------

        occupancy_grid[
            gy,
            gx
        ] += OCCUPIED_UPDATE


    # Keep values bounded
    np.clip(
        occupancy_grid,
        LOG_ODDS_MIN,
        LOG_ODDS_MAX,
        out=occupancy_grid
    )


# ============================================================
# BUILD LOCAL OBSTACLE SEGMENTS
# ============================================================

def build_segments(
    angles_deg,
    distances_mm
):

    if len(angles_deg) < 2:
        return []

    distances_m = distances_mm / 1000.0

    valid = (
        np.isfinite(distances_m)
        & (distances_m > MIN_DISTANCE_METERS)
        & (distances_m <= MAX_DISTANCE_METERS)
    )

    angles_deg = angles_deg[valid]
    distances_m = distances_m[valid]

    if len(angles_deg) < 2:
        return []

    # Reduce points for rendering
    if len(angles_deg) > MAX_SCAN_POINTS:

        indices = np.linspace(
            0,
            len(angles_deg) - 1,
            MAX_SCAN_POINTS
        ).astype(int)

        angles_deg = angles_deg[indices]
        distances_m = distances_m[indices]


    local_x, local_y = lidar_to_local(
        angles_deg,
        distances_m * 1000
    )

    points = np.column_stack([
        local_x,
        local_y
    ])

    segments = []

    for i in range(len(points) - 1):

        p1 = points[i]
        p2 = points[i + 1]

        gap = np.linalg.norm(
            p2 - p1
        )

        if gap <= CONNECT_DISTANCE:

            segments.append([
                p1,
                p2
            ])

    return segments


# ============================================================
# MAP DISPLAY UPDATE
# ============================================================

def update_map_display():

    # Occupancy probability
    probability = (
        1
        -
        1 /
        (
            1
            + np.exp(
                occupancy_grid
            )
        )
    )

    map_image.set_data(
        probability
    )


# ============================================================
# MAIN PLOT UPDATE
# ============================================================

def update_plot(frame):

    global last_mapped_scan

    # --------------------------------------------------------
    # Get current data
    # --------------------------------------------------------

    with data_lock:

        current_x = robot_x
        current_y = robot_y
        current_heading = robot_heading

        current_scan_counter = scan_update_counter

    angles, distances = get_combined_scan()


    # --------------------------------------------------------
    # Nothing received yet
    # --------------------------------------------------------

    if len(angles) == 0:

        return (
            local_environment,
            local_environment_glow,
            local_scan_points,
            robot_marker,
            robot_direction,
            robot_path,
            global_robot_marker
        )


    # ========================================================
    # LOCAL LiDAR DISPLAY
    # ========================================================

    distances_m = distances / 1000.0

    valid = (
        np.isfinite(angles)
        & np.isfinite(distances_m)
        & (distances_m > MIN_DISTANCE_METERS)
        & (distances_m <= MAX_DISTANCE_METERS)
    )

    valid_angles = angles[valid]
    valid_distances = distances_m[valid]

    local_x, local_y = lidar_to_local(
        valid_angles,
        valid_distances * 1000
    )


    # Update scan points
    local_scan_points.set_offsets(
        np.column_stack([
            local_x,
            local_y
        ])
    )


    # Update connected surfaces

    segments = build_segments(
        valid_angles,
        valid_distances * 1000
    )

    local_environment.set_segments(
        segments
    )

    local_environment_glow.set_segments(
        segments
    )


    # ========================================================
    # UPDATE GLOBAL MAP
    # ========================================================

    if current_scan_counter != last_mapped_scan:

        last_mapped_scan = current_scan_counter

        # Downsample mapping scan
        if len(local_x) > MAX_SCAN_POINTS:

            indices = np.linspace(
                0,
                len(local_x) - 1,
                MAX_SCAN_POINTS
            ).astype(int)

            map_local_x = local_x[indices]
            map_local_y = local_y[indices]

        else:

            map_local_x = local_x
            map_local_y = local_y


        update_occupancy_map(
            map_local_x,
            map_local_y,
            current_x,
            current_y,
            current_heading
        )

        update_map_display()


    # ========================================================
    # UPDATE ROBOT TRAJECTORY
    # ========================================================

    robot_path_x.append(current_x)
    robot_path_y.append(current_y)

    # Prevent excessive trajectory storage
    if len(robot_path_x) > 5000:

        del robot_path_x[:-5000]
        del robot_path_y[:-5000]


    robot_path.set_data(
        robot_path_x,
        robot_path_y
    )


    # ========================================================
    # ROBOT LOCAL VIEW
    # ========================================================

    robot_marker.set_data(
        [0],
        [0]
    )

    heading = np.radians(
        current_heading
    )

    direction_length = 0.5

    direction_x = (
        direction_length
        * np.sin(heading)
    )

    direction_y = (
        direction_length
        * np.cos(heading)
    )

    robot_direction.set_data(
        [0, direction_x],
        [0, direction_y]
    )


    # ========================================================
    # GLOBAL ROBOT LOCATION
    # ========================================================

    global_robot_marker.set_data(
        [current_x],
        [current_y]
    )


    # ========================================================
    # STATUS
    # ========================================================

    status_text.set_text(
        f"● LIVE    "
        f"X: {current_x:.2f} m    "
        f"Y: {current_y:.2f} m    "
        f"Heading: {current_heading:.1f}°"
    )


    # ========================================================
    # OBSTACLE WARNING
    # ========================================================

    close_obstacles = (
        valid_distances
        <
        OBSTACLE_WARNING_DISTANCE
    )

    if np.any(close_obstacles):

        obstacle_warning.set_text(
            "⚠ OBSTACLE DETECTED"
        )

    else:

        obstacle_warning.set_text(
            "✓ PATH CLEAR"
        )


    return (
        local_environment,
        local_environment_glow,
        local_scan_points,
        robot_marker,
        robot_direction,
        robot_path,
        global_robot_marker,
        map_image,
        status_text,
        obstacle_warning
    )

# MAIN

        "Lidar"

        value_changed_callback

        value_changed_callback

        "Launching LiDAR localisation + mapping system..."

    # ========================================================

    # --------------------------------------------------------

        linewidth=1.5,
        alpha=0.8

    # Global robot marker
    # --------------------------------------------------------

        [],

        marker="o",
        markersize=9,
        markerfacecolor="#39FF14",
        markeredgecolor="white",
        markeredgewidth=1.5,
        linestyle="None",
        zorder=20
    )


    # ========================================================
    # LOCAL LiDAR VIEW
    # ========================================================

    ax_local = fig.add_subplot(122)

    ax_local.set_title(
        "LIVE LOCAL LiDAR",
        color="#39FF14",
        fontsize=15,
        fontweight="bold"
    )

    ax_local.set_facecolor(
        "#050705"
    )

    ax_local.set_aspect("equal")

    ax_local.set_xlim(
        -MAX_DISTANCE_METERS,
        MAX_DISTANCE_METERS
    )

    ax_local.set_ylim(
        -MAX_DISTANCE_METERS,
        MAX_DISTANCE_METERS
    )

    ax_local.set_xlabel(
        "Right / Left (m)",
        color="#6f8f78"
    )

    ax_local.set_ylabel(
        "Forward / Back (m)",
        color="#6f8f78"

        colors="#6f8f78"

        color="#18351f",

    ax_local.add_collection(

    ani = FuncAnimation(
