import time
import threading
import numpy as np
import matplotlib.pyplot as plt
from matplotlib.animation import FuncAnimation
from networktables import NetworkTables
from collections import deque

import logging
logging.basicConfig(level=logging.WARNING)

# --- CONFIGURATION ---
ROBOT_IP = '10.12.34.2' # Change this to your robot's IP
MAX_DISTANCE_METERS = 5.0

data_lock = threading.Lock()

# --- CONTINUOUS ROLLING BUFFER ---
HISTORY_LENGTH = 1500
rolling_angles = deque(maxlen=HISTORY_LENGTH)
rolling_distances = deque(maxlen=HISTORY_LENGTH)

# Temporary sync buffers for multi-part network data
temp_angles_1 = []
temp_distances_1 = []
temp_angles_2 = []
temp_distances_2 = []

def value_changed_callback(table, key, value, isNew):
    global temp_angles_1, temp_distances_1, temp_angles_2, temp_distances_2
    global rolling_angles, rolling_distances
    
    with data_lock:
        if key == 'ScanAngles_Part1':
            temp_angles_1 = value
        elif key == 'ScanDistances_Part1':
            temp_distances_1 = value
            min_len = min(len(temp_angles_1), len(temp_distances_1))
            if min_len > 0:
                rolling_angles.extend(temp_angles_1[:min_len])
                rolling_distances.extend(temp_distances_1[:min_len])
                
        elif key == 'ScanAngles_Part2':
            temp_angles_2 = value
        elif key == 'ScanDistances_Part2':
            temp_distances_2 = value
            min_len = min(len(temp_angles_2), len(temp_distances_2))
            if min_len > 0:
                rolling_angles.extend(temp_angles_2[:min_len])
                rolling_distances.extend(temp_distances_2[:min_len])

def update_plot(frame):
    with data_lock:
        if len(rolling_angles) == 0:
            return scatter,
        
        angles_deg = np.array(rolling_angles)
        distances_mm = np.array(rolling_distances)
    
    distances_m = distances_mm / 1000.0
    angles_rad = np.radians(angles_deg)
    
    # Filter invalid points
    valid_points = distances_m > 0.05 
    
    scatter.set_offsets(np.c_[angles_rad[valid_points], distances_m[valid_points]])
    return scatter,

def main():
    print(f"Connecting to NetworkTables at {ROBOT_IP}...")
    NetworkTables.initialize(server=ROBOT_IP)
    NetworkTables.setUpdateRate(0.010)
    
    lidar_table = NetworkTables.getTable("Lidar")
    lidar_table.addEntryListener(value_changed_callback)

    global scatter
    fig = plt.figure(figsize=(8, 8))
    
    ax = fig.add_subplot(111, projection='polar')
    ax.set_title("Live Studica Lidar Stream (Full 360 Map - Unreduced)", va='bottom', color='white')
    
    ax.set_theta_zero_location("N")
    ax.set_theta_direction(-1)
    ax.set_ylim(0, MAX_DISTANCE_METERS)
    
    scatter = ax.scatter([], [], c='#39FF14', s=15, alpha=0.75, edgecolors='black')
    
    fig.patch.set_facecolor('#1e1e1e')
    ax.set_facecolor('#000000')
    ax.tick_params(colors='white')
    ax.grid(color='#333333')
    ax.spines['polar'].set_color('#555555')

    ani = FuncAnimation(fig, update_plot, interval=50, blit=True, save_count=50)
    
    print("Launching full 360 radar viewer...")
    plt.show()

if __name__ == "__main__":
    main()
